  TooltipProvider,

  TooltipContent,
} from "@/components/ui/tooltip";
import { Button } from "@/components/ui/button";

export const tooltipContentProps = [
  {
    name: "side",
    type: '"top" | "right" | "bottom" | "left"',
    default: '"top"',
    description: "ツールチップの表示位置",
  },
  {
    name: "sideOffset",
    type: "number",
    default: "4",
    description: "トリガーからの表示位置のオフセット（px）",
  },
];

export const tooltipProviderProps = [
  {
    name: "delayDuration",

# Tooltip

<Description>要素にホバーすると表示される補足情報のポップアップ。</Description>

## Installation

<CodeBlock
  code={`npx shadcn@latest add https://kasumi-ui.vercel.app/r/tooltip.json`}
/>

## Usage

<ComponentPreview>
  <TooltipProvider>
    <Tooltip>
      <TooltipTrigger asChild>
        <Button variant="outline">ホバーしてみて</Button>
      </TooltipTrigger>
      <TooltipContent>
        <p>補足テキスト</p>
      </TooltipContent>
    </Tooltip>
  </TooltipProvider>
</ComponentPreview>

<CodeBlock code={`import {
  TooltipProvider,
  Tooltip,
  TooltipTrigger,
  TooltipContent,
} from "@/components/ui/tooltip";
import { Button } from "@/components/ui/button";

<TooltipProvider>
  <Tooltip>
    <TooltipTrigger asChild>
      <Button variant="outline">ホバーしてみて</Button>
    </TooltipTrigger>
    <TooltipContent>
      <p>補足テキスト</p>
    </TooltipContent>
  </Tooltip>
</TooltipProvider>`} />
